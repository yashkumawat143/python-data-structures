1....Count even and odd numbers in a list.
answer : 

n1=input("Enter the numbers : ");
answer=n1.split();
countodd=0;
counteven=0;
for i in answer:
    if(int(i)%2==0):
        counteven+=1;
    else:
        countodd+=1;
print("Total even number = ",counteven);
print("Total odd number =",countodd);


2....Find average of list elements.
answer : 

n1=input("Enter the number : ");
answer=n1.split();
sum=0;
for i in answer:
    sum=sum+int(i);
    
average=(sum)/len(answer);
print("The average of number = ",average);    



3....Find largest number in a list.
answer : 

n1=input("Enter the number : ");
answer=n1.split();
largest="";
for i in answer:
    if(i>largest):
        largest=i;
print("The largest number = ",largest);

7....Remove duplicates from a list.
answer : 

n1=input("Enter the elements : ");
answer=n1.split();
num=[];
for i in answer:
    if(i not in num):
        num.append(i);
print(num);

9....Find common elements between two lists.
answer : 

n1=input("Enter the words : ");
n2=input("Enter the words : ");
answer1=n1.split();
answer2=n2.split();
num=[];
for i in answer1:
    for j in answer2:
        if(i==j):
            num.append(i);
print("The commom word = ",num);
