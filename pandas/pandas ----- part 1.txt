#based on single condition

import pandas as pd

data={
    "name":["yash","zoro","Obito","luffy"],
    "age":[12,34,45,56],
    "salary":[2000,30000,60000,54000]
}
a=pd.DataFrame(data);
print(a);
print("The salray grater then 30000");
new=a[a["salary"]>30000];
print(new);





#combine multiple condition.

import pandas as pd

data={
    "name":["yash","zoro","Obito","luffy"],
    "age":[12,34,45,56],
    "salary":[2000,30000,60000,54000]
}
a=pd.DataFrame(data);
print(a);
print("The salray grater then 30000 and also age is greator then 34");
new=a[(a["salary"]>30000)&(a["age"]>34)];
print(new);




# or condition.

import pandas as pd

data={
    "name":["yash","zoro","Obito","luffy"],
    "age":[12,34,45,56],
    "salary":[2000,30000,60000,54000]
}
a=pd.DataFrame(data);
print(a);
print("The age is greator then 34 or the salary greator then 30000");
new=a[(a["age"]>34)|(a["salary"]>30000)];
print(new);
