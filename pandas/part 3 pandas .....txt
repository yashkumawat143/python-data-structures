import pandas as pd

    "age":[10,20,8],

}

#Grouping.


import pandas as pd

data={
    "name":['arun','tarun','anuj'],
    "age":[10,20,8],
    "salary":[200,377000,47000]
}
a=pd.DataFrame(data);

group=a.groupby("age")["salary"].sum();
print(group);

#merging and joining the data.


----->>>>>

import pandas as pd

customer={
       "id":[1,2,3,4],
       "name":['yash','hello','tanu','kannu']
}
order={
    'id':[1,2,3,6],
    'orderamount':[12,23,34,45]
}
s=pd.DataFrame(customer);
s1=pd.DataFrame(order);
merg=pd.merge(s,s1,on='id',how='inner');
print(merg);

#verticaly.


import pandas as pd

customer={
       "id":[1,2,3,4],
       "name":['yash','hello','tanu','kannu']
}
order={
    'id':[1,2,3,6],
    'name':['pri','sunil','tannnu','kannnnu']
}
s=pd.DataFrame(customer);
s1=pd.DataFrame(order);

con=pd.concat([s,s1],axis=0,ignore_index=True);
print(con);
