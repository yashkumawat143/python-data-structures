1....Find the maximum element.
answer :

n1=input("Enter the numbers : ");
new=list(map(int,n1.split()));
arr=array('i',new);
max=arr[0];
for i in arr:
    if(i>max):
        max=i;
print(max);

15....Remove duplicate elements.
answer :

from array import *

n1=input("Enter the number : ");
new=list(map(int,n1.split()));
arr=array('i',new);
a=[];
for i in arr:
    if(i not in a):
        a.append(i);
print(a);
