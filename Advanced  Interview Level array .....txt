1....Check if an array is sorted.
answer :

from array import *

n1=input("Enter the number : ");
new=list(map(int,n1.split()));
arr=array('i',new);

if(list(arr)==sorted(arr)):
    print("It is sorted ");
else:
    print("It is not sorted");

answer :

3....Find the duplicate element in an array.
answer :

from array import *

n1=input("Enter the number : ");
new=list(map(int,n1.split()));
arr=array('i',new);
a=[];
for i in arr:
    if(i in a):
        print(i);
    else:
        a.append(i);



4....Move all zeros to the end.
answer :

from array import *

n1=input("Enter the number : ");
new=list(map(int,n1.split()));
arr=array('i',new);

j=0;
for i in range(len(arr)):
    if(arr[i]!=0):
        arr[i],arr[j]=arr[j],arr[i];
        j+=1;
print(arr);

8....Find the kth largest element.
answer :

from array import *

n1=input("Enter the number : ");
k=int(input("Enter the k :"))
new=list(map(int,n1.split()));
arr=array('i',new);
a=sorted(arr);
j=a[::-1];
print("The largest element : ",j[k-1]);
