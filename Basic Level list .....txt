12....Sort a list in descending order.
answer : 

n1=input("Enter the words : ");

13....Find maximum element.
answer :

n1=input("Enter the words : ");
answer=n1.split();
largest="";
for i in answer:
    if(len(i)>len(largest)):
        largest=i;
print(largest);


14....Find minimum element.
answer : 

n1=input("Enter the words : ");
answer=n1.split();
smallest=answer[0];
for i in answer:
    if(len(i)<len(smallest)):
        smallest=i;
print(smallest);
